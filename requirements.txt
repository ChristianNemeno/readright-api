# REPRODUCIBILITY: for deployment, generate a full lockfile from a known-good,
# CI-tested Python 3.12 venv and build from that instead of this loose list:
#     pip install -r requirements.txt && pip freeze > requirements.lock.txt
# Then point the Dockerfile at requirements.lock.txt. The pins below only fix the
# few axes that are known to break WhisperX builds; everything else floats.
#
# CPU-ONLY TORCH: this service runs on CPU (WHISPERX_DEVICE=cpu, hardcoded). The
# index below serves CPU wheels for torch/torchaudio/torchvision first, so the
# image doesn't pull the ~4.6GB of unused nvidia-cu12-*/CUDA-torch packages that
# whisperx/pyannote otherwise bring in. Comment it out only for a deliberate GPU build.
# ─────────────────────────────────────────────────────────────────────────────

# GO2 — ASR pipeline (heavy)
# NOTE: ffmpeg system binary required (installed via the Dockerfile / apt).
# whisperx and ctranslate2 are pinned, not just floated: an unpinned resolve backtracks
# pip all the way down to whisperx==3.3.1, which drags in ctranslate2==4.4.0 — a build
# whose shared library fails to even import ("cannot enable executable stack as shared
# object requires"). Confirmed broken; do not loosen these two without re-verifying
# `import whisperx` actually works on a clean install first.
whisperx==3.8.6
ctranslate2==4.8.1   # int8 inference backend for the model (see WHISPERX_COMPUTE_TYPE)
faster-whisper       # used to pre-download/quantize the model at image-build time
librosa
praat-parselmouth
ffmpeg-python
numpy>=2.1.0         # whisperx==3.8.6 requires numpy>=2.1.0 — do not pin <2, it forces pip to
                     # backtrack to whisperx<3.4, which drags in a broken ctranslate2 build
